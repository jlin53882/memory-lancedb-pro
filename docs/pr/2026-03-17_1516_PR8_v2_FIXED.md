### 2.4 DataTable 分頁設計

```python
# 每頁顯示數量
RULES_PAGE_SIZE = 50

class RulesMainView(ft.Container):
    def __init__(self, page):
        # 分頁狀態
        self._current_page = 0
        self._total_pages = 0
        self._all_rules = []  # 快取全部規則
        
        # 分頁控制
        self._pagination = ft.Row([
            ft.TextButton("上一頁", on_click=self._prev_page),
            ft.Text(f"第 {self._current_page + 1} / {self._total_pages} 頁"),
            ft.TextButton("下一頁", on_click=self._next_page),
        ])
    
    def _load_page(self, page: int):
        """分頁載入規則"""
        start = page * RULES_PAGE_SIZE
        rules = self._all_rules[start:start + RULES_PAGE_SIZE]
        self._rules_table.rows = self._build_table_rows(rules)
        self.update()
```
