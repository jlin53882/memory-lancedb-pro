```javascript
// Test 1: Phase-2 確認 lock 次數 = 1 per batch
async function testPhase2_LockCountFixed() {
  let lockCount = 0;
  const store = {
    async list() { return Array.from({ length: 10 }, (_, i) => createLegacyEntry(`entry-${i}`, `Memory ${i}`)); },
    async update(id, patch) { return true; },
    async runWithFileLock(fn) { lockCount++; return fn(); },
  };
  
  const upgrader = createMemoryUpgraderPhase2(store, llm, { log: () => {} });
  await upgrader.upgrade({ batchSize: 10, noLlm: true });
  
  assert.equal(lockCount, 1, "Phase-2 應該只有 1 次 lock");
}

// Test 7: 舊實作 vs 新實作比較
async function testCompareOldVsNew() {
  // 舊實作：每個 entry 拿一次 lock
  let oldLockCount = 0;
  for (const entry of oldMemories) {
    oldLockCount++; // 每個 entry 都拿 lock
    await oldStore.update(entry.id, { text: "upgraded" });
  }
  
  // 新實作：一次 lock 寫入整批
  await newUpgrader.upgrade({ batchSize: 5, noLlm: true });
  
  console.log(`  舊實作 Lock 次數: ${oldLockCount} (每個 entry 1 次)`);
  console.log(`  新實作 Lock 次數: ${newLockCount} (每個 batch 1 次)`);
  console.log(`  改善: ${oldLockCount - newLockCount} 次 lock`);
}
```
